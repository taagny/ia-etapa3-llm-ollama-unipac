```python
import unicodedata

def é_palíndromo(s):
    # Limpa a string
    cleaned_s = ''.join(c.lower()
                        for c in unicodedata.normalize('NFD', s)
                        if unicodedata.category(c) in ('Ll', 'Lv', 'Lo')
                       )
    cleaned_s = cleaned_s.replace(' ', '')
    
    # Verifica se a string é um palíndromo
    return cleaned_s == cleaned_s[::-1]

# Testes utilizando o assert
```
